de <stdio.h>

int main(void)
{
    int a[3][3] =
    {
        {1, 2, 3},
        {4, 5, 6},
        {7, 8, 9}
    };

    int i, j;
    int sum;

    for (j = 0; j < 3; j++)
    {
        sum = 0;

        for (i = 0; i < 3; i++)
        {
            sum = sum + a[i][j];
        }

        printf("sum (column %d) = %d\n", j + 1, sum);
    }

    return 0;
}
